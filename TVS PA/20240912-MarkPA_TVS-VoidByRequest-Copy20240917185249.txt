-- Step 1: Access to TESTDB061 (10.18.200.56) -> bodb02 and run below script
Declare 	@custid int = 1053056; -- input CustID

Declare 	@SubIssueTypeId tinyint = 28, -- input Reason/PA (data in file document)
		@IssueTypeId tinyint = 11; -- 11/13

Declare 
		@ActionId tinyint = 1,
		@DecidedBy nvarchar = 'starix3',
		@Remark nvarchar = 'qc test',
		@UserId int = 169368,
		@CreatedDate datetime = getdate(),
		@LastModifiedBy int = 169368,
		@LastModifiedDate  datetime = getdate(),
		@IsTest bit = 0,
		@MatchId int,
		@LeagueId int,
		@HomeId int,
		@AwayId int,
		@RequestType tinyint=0,
		@Sporttype smallint,
		@TransId bigint,
		@Bettype smallint,
		@BetId bigint,
		@SiteDBGroupID smallint = 3,
		@IsProcessed tinyint = 1;

SELECT top 1 @MatchId = m.matchid, -- 29: Parlay, m.matchid: NOT Parlay
			@TransId = b.TransId, 
			@LeagueId = m.leagueid,
			@HomeId = m.homeid,
			@AwayId = m.awayid,
			@Sporttype = m.sporttype,
			@Bettype = b.bettype,
			@BetId = b.betid
FROM dbo.match m 
INNER JOIN bettrans b 
ON b.matchid = m.matchid
WHERE m.eventdate <= getdate() 
-- and m.sporttype=1
order by NEWID();

INSERT INTO [bodb02].[dbo].[TicketVoiding_Request] (ActionId,DecidedBy,Remark,UserId,CreatedDate,LastModifiedBy,LastModifiedDate,IsTest,IssueTypeId) Values
(@ActionId,@DecidedBy,@Remark,@UserId,@CreatedDate,@LastModifiedBy,@LastModifiedDate,@IsTest,@IssueTypeId);

Declare @RequestId int = (select top 1 (RequestId) from  [bodb02].[dbo].[TicketVoiding_Request] where UserId = 169368 order by CreatedDate desc);

INSERT INTO [bodb02].[dbo].[TicketVoiding_RequestDetail]  (RequestId,MatchId,LeagueId,HomeId,AwayId,RequestType,SubIssueTypeId,Sporttype) values
(@RequestId,@MatchId,@LeagueId,@HomeId,@AwayId,@RequestType,@SubIssueTypeId,@Sporttype);

Declare @RequestDetailId int = (select top 1 ID from  [bodb02].[dbo].[TicketVoiding_RequestDetail] where requestid = @RequestId);

INSERT INTO [bodb02].[dbo].[TicketVoiding_RequestDetailTicket_Single] (RequestId,IssueTypeId,SubIssueTypeId,TransId,MatchId,RequestDetailId,Bettype,BetId,CustId,SiteDBGroupID,IsProcessed) values
(@RequestId,@IssueTypeId,@SubIssueTypeId,@TransId,@MatchId,@RequestDetailId,@Bettype,@BetId,@CustId,@SiteDBGroupID,@IsProcessed);

SELECT TOP (1000) *  FROM [bodb02].[dbo].[TicketVoiding_Request] WHERE RequestId = @RequestId;
SELECT TOP (1000) *  FROM [bodb02].[dbo].[TicketVoiding_RequestDetail] WHERE RequestId = @RequestId;
select TOP (1000) *  FROM [bodb02].[dbo].[TicketVoiding_RequestDetailTicket_single] WHERE RequestId = @RequestId;
