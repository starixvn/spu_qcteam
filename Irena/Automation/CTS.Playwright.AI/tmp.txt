FROM mcr.microsoft.com/playwright:v1.54.0-jammy

# Set working directory
WORKDIR /app

# Define environment variables with defaults
ENV ENV=local \
    BROWSER=chrome \
    MODE=''

# Copy package files
COPY package*.json ./

# Install dependencies
RUN npm ci && npm install dotenv

# Install selected browsers
RUN npx playwright install chromium firefox msedge --with-deps

# Create cache directories
RUN mkdir -p ~/.cache/ms-playwright/

# Copy remaining files
COPY . .

# Cache the Playwright browser binaries
VOLUME ["~/.cache/ms-playwright/"]

# Create a startup script
RUN echo '#!/bin/sh\nnode runner.js ENV=${ENV} BROWSER=${BROWSER} MODE=${MODE}' > /app/start.sh && \
    chmod +x /app/start.sh

# Set default command to use the startup script
CMD ["/app/start.sh"]
