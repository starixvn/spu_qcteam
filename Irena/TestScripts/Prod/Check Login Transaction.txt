-- Desktop, Mb Web: IP, Device
select count(distinct IP) as IP from DCS_DataCenter.AccountIP where AccountID = 110993536 and LastTransDate between DATE_SUB(CURDATE(), INTERVAL 90 DAY) and CURDATE() and IP <> ''; -- 4

-- Mb App: Bot, Invalid, Total Login
select sum(case when FlaggedGroupID = 3 then TotalTrans else 0 end) as MBBot, -- 0
       sum(case when FlaggedGroupID = 2 then TotalTrans else 0 end) as MBInvalid -- 0
from DCS_DataTrace.MBLoginTransactionSummary as l 
where MBAccountID = 2228 and TransDate between GREATEST('2025-07-18 05:37:09.9400000
',(DATE_SUB(CURDATE(), INTERVAL 90 DAY))) and CURDATE();

select count(id) as MBTotalLogin from  DCS_DataCenter.MBTransaction07 -- 2421
where MBAccountID = 2228 and TransTime >= GREATEST('2025-07-18 05:37:09.9400000
',(DATE_SUB(CURDATE(), INTERVAL 90 DAY))) and TransTime < DATE_SUB(CURDATE(), INTERVAL -1 DAY);

-- Desktop, Mb Web: Bot, Invalid, Total Login, Desktop, Mb Web
select sum(case when FlaggedGroupID = 3 then TotalTrans else 0 end) as Bot, -- 0
       sum(case when FlaggedGroupID = 2 then TotalTrans else 0 end) as Invalid, -- 1
       sum(TotalTrans) as TotalLogin -- % Bot, % Invalid Browser
from DCS_DataTrace.LoginTransactionSummary as l 
where AccountID = 110993536 and TransDate between DATE_SUB(CURDATE(), INTERVAL 90 DAY) and CURDATE();

select count(case when dt.GroupName = 'Desktop' then 1 else null end) as Desktop,-- 0
    count(case when dt.GroupName = 'Mobile' then 1 else null end) as MobileWeb,-- 5
    count(1) as TotalLogin-- 5
from DCS_DataCenter.Transaction07 as t

    left join DCS_DataCenter.DeviceType as dt on dt.DeviceTypeID = ug.DeviceTypeID
    where t.AccountID = 110993536 and t.CreatedDate >= DATE_ADD('2025-10-20', INTERVAL 1 DAY);
