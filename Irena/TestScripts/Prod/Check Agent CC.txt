/* Server - DB: cts - CTS_DataCenter */

-- Get All Agent
select distinct cus.CustID, cus.Username, cus.Site, cus.CreatedDate
from CTS_DataCenter.CTSCustomer as cus
where cus.RoleID = 2 and cus.CustSubID = 0 and cus.CreatedDate <= '2024-11-22 23:59:59'
order by cus.CustID;
-- Get MAX LastTicketDate
with cteAgent as (
select ag.CustID AS AgentCustID, ag.CTSCustID AS AgentCTSCustID, MAX(ass.LastTicketDate) AS MAXLastTicketDate
from CTS_DataCenter.CTSCustomer as ag
    inner join CTS_DataCenter.CTSCustomer as me on ag.CustID = me.recommend   
    left join CTS_Archive.CTSCustomerAssociationStatus as ass on me.CTSCustID = ass.CTSCustID and me.RoleID = 1
where ag.RoleID = 2 and ag.CustSubID = 0 and ag.CreatedDate <= '2024-11-22 23:59:59'
and ag.CustID between 51464328 and 59601121 --59601122 67168910, 
GROUP BY AgentCustID, AgentCTSCustID
)
select t.AgentCustID, t.AgentCTSCustID, t.MAXLastTicketDate
from cteAgent as t
order by t.AgentCustID;
-- Get Category-CC
select cus.CustID, GROUP_CONCAT(cc.CategoryName), GROUP_CONCAT(cc.CustomerClass), MIN(agCC.CreatedDate), MIN(agCC.LastScannedDate)
from CTS_DataCenter.CTSCustomer as cus
    inner join CTS_DataCenter.CTSCustomerClassificationAgency as agCC on agCC.CTSCustID = cus.CTSCustID 
    inner join CTS_DataCenter.CustomerCategoryAgency as cc on cc.ParentID = agCC.ParentID and cc.CategoryID = agCC.CategoryID
where cus.RoleID = 2 and cus.CustSubID = 0 and cus.CreatedDate <= '2024-11-22 23:59:59'
group by cus.CustID;

with cteAgent as (
SELECT CustID, MAX(CreatedTime) AS MaxCreatedTime
FROM CTS_DataCenter.CTSCustomerClassificationAgency
where CustID IN (1265)
group by CustID
) 
select cc.*
from cteAgent as t
inner join CTS_DataCenter.CTSCustomerClassificationAgency as cc on cc.CustID = t.CustID and cc.CreatedTime = t.MaxCreatedTime;
