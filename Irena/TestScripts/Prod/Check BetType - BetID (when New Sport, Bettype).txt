select * from dbo.league where sporttype = 5 and LeagueGroupID not in (42,72,74,108,113,122) and leaguename not like '%set%' and leaguename not like '%winner%';
select * from dbo.team where sporttype = 5 and teamname not like '%set%' and teamname not like '%winner%';

select SportType, BetType, IsLive 
from dbo.SportBettypeMatrix 
where SportType=5 and BetType in (153, 154, 155, 156);

select b.BetType, b.BetID, b.Betteam, COUNT(1) 
from dbo.bettrans  AS b with(nolock)
  inner join dbo.match AS m with(nolock) on m.matchid = b.matchid
where m.SportType=5 and b.BetType in (153, 154, 155, 156) 
group by b.BetType, b.BetID, b.Betteam
order by b.Bettype ASC;
